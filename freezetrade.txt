local halfW = 135
local StealBtn, StealCircle = createToggle("Freeze Screen", 16, 124, halfW)
local AcceptBtn, AcceptCircle = createToggle("Auto Accept", 159, 124, halfW)

local function setToggle(btn, circle, state)
    if state then
        TweenService:Create(circle, TweenInfo.new(0.18), {Position = UDim2.new(1, -19, 0.5, -8), BackgroundColor3 = Color3.fromRGB(255, 255, 255)}):Play()
        TweenService:Create(btn, TweenInfo.new(0.18), {BackgroundColor3 = Color3.fromRGB(255, 255, 255)}):Play()
    else
        TweenService:Create(circle, TweenInfo.new(0.18), {Position = UDim2.new(0, 3, 0.5, -8), BackgroundColor3 = Color3.fromRGB(60, 60, 60)}):Play()
        TweenService:Create(btn, TweenInfo.new(0.18), {BackgroundColor3 = Color3.fromRGB(30, 30, 30)}):Play()
    end
end

-- USERNAME LOOKUP
-- =====================================================
local function resetProfile()
    ProfilePic.Image = ""
    StatusIndicator.Text = "?"
    StatusIndicator.TextColor3 = Color3.fromRGB(60, 60, 60)
    StatusIndicator.Visible = true
    PicStroke.Color = Color3.fromRGB(50, 50, 50)
    targetUserId = nil
    targetUsername = ""
    if stealEnabled then stealEnabled = false setToggle(StealBtn, StealCircle, false) end
    if acceptEnabled then acceptEnabled = false setToggle(AcceptBtn, AcceptCircle, false) end
end

local function lookupUsername(username)
    if username == "" then resetProfile() return end
    StatusIndicator.Text = "..."
    StatusIndicator.TextColor3 = Color3.fromRGB(180, 180, 60)
    StatusIndicator.Visible = true
    ProfilePic.Image = ""

    task.spawn(function()
        local ok, userId = pcall(function()
            return Players:GetUserIdFromNameAsync(username)
        end)
        if UsernameBox.Text:lower() ~= username:lower() then return end
        if ok and userId then
            targetUserId = userId
            targetUsername = username
            ProfilePic.Image = "https://www.roblox.com/headshot-thumbnail/image?userId=" .. userId .. "&width=150&height=150&format=png"
            PicStroke.Color = Color3.fromRGB(80, 200, 80)
            StatusIndicator.Visible = false
        else
            resetProfile()
            StatusIndicator.Text = "X"
            StatusIndicator.TextColor3 = Color3.fromRGB(200, 50, 50)
            PicStroke.Color = Color3.fromRGB(200, 50, 50)
        end
    end)
end

UsernameBox:GetPropertyChangedSignal("Text"):Connect(function()
    local current = UsernameBox.Text
    if current == "" then resetProfile() return end
    task.delay(0.6, function()
        if UsernameBox.Text == current then lookupUsername(current) end
    end)
end)

-- =====================================================
-- DRAG SYSTEM
-- =====================================================
local dragging, dragInput, dragStart, startPos = false, nil, nil, nil

TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and input == dragInput then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
